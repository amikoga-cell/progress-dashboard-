  <script>

    var GIST_ID = '2bf253d3446c8999d088826891a2073c';

    var STATUSES = ['未着手', '弊社ボール', '先方待ち', '完了'];

    var TASKS = ['発行枚数確定', 'デザイン数確定', '見積もり送付', '発注書締結', '商品登録', 'URL発行・共有', 'デザイン共有', '印刷会社へ発注', '納品'];

    var allData = [];

    var showCompleted = false;

    var token = '';


    function init() {

      token = localStorage.getItem('gh_token') || '';

      if (!token) {

        document.getElementById('tokenModal').classList.add('show');

        return;

      }

      loadData();

    }


    function saveToken() {

      var t = document.getElementById('tokenInput').value.trim();

      if (!t) { alert('トークンを入力してください'); return; }

      localStorage.setItem('gh_token', t);

      token = t;

      document.getElementById('tokenModal').classList.remove('show');

      loadData();

    }


    function loadData() {

      document.getElementById('loading').textContent = '読み込み中...';

      fetch('https://api.github.com/gists/' + GIST_ID, {

        headers: { 'Authorization': 'token ' + token }

      })

      .then(function(r) {

        if (!r.ok) throw new Error('認証エラー。トークンを確認してください。');

        return r.json();

      })

      .then(function(data) {

        var content = data.files['data.json'].content;

        var parsed = JSON.parse(content);

        allData = parsed.companies || [];

        render();

        document.getElementById('loading').style.display = 'none';

        document.getElementById('dashTable').style.display = 'table';

        updateTime();

      })

      .catch(function(e) {

        document.getElementById('loading').textContent = 'エラー: ' + e.message;

        localStorage.removeItem('gh_token');

      });

    }


    function saveData() {

      return fetch('https://api.github.com/gists/' + GIST_ID, {

        method: 'PATCH',

        headers: {

          'Authorization': 'token ' + token,

          'Content-Type': 'application/json'

        },

        body: JSON.stringify({

          files: {

            'data.json': {

              content: JSON.stringify({ companies: allData }, null, 2)

            }

          }

        })

      });

    }


    function isAllDone(company) {

      return TASKS.every(function(t) { return company.tasks[t] === '完了'; });

    }


    function render() {

      renderStats();

      renderTable();

    }


    function renderStats() {

      var active = allData.filter(function(c) { return !isAllDone(c); });

      var ourBall = 0, waiting = 0;

      active.forEach(function(c) {

        TASKS.forEach(function(t) {

          if (c.tasks[t] === '弊社ボール') ourBall++;

          if (c.tasks[t] === '先方待ち') waiting++;

        });

      });

      var doneCount = allData.filter(isAllDone).length;


      document.getElementById('statsArea').innerHTML =

        '<div class="stat-card"><div class="num num-弊社ボール">' + ourBall + '</div><div class="label">弊社ボール</div></div>' +

        '<div class="stat-card"><div class="num num-先方待ち">' + waiting + '</div><div class="label">先方待ち</div></div>' +

        '<div class="stat-card"><div class="num num-完了社">' + doneCount + '</div><div class="label">完了事業者</div></div>';

    }


    function renderTable() {

      var headHtml = '<tr><th>事業者名</th>';

      TASKS.forEach(function(t) { headHtml += '<th>' + t + '</th>'; });

      headHtml += '</tr>';

      document.getElementById('tableHead').innerHTML = headHtml;


      var filtered = allData.filter(function(c) { return showCompleted || !isAllDone(c); });

      var bodyHtml = '';


      filtered.forEach(function(c) {

        bodyHtml += '<tr class="' + (isAllDone(c) ? 'company-done' : '') + '">';

        bodyHtml += '<td>' + esc(c.name) + '</td>';

        TASKS.forEach(function(t) {

          var s = c.tasks[t] || '未着手';

          bodyHtml += '<td><span class="badge badge-' + s + '" onclick="cycleStatus(\'' + escAttr(c.name) + '\',\'' + escAttr(t) + '\',this)">' + s + '</span></td>';

        });

        bodyHtml += '</tr>';

      });


      if (!filtered.length) {

        bodyHtml = '<tr><td colspan="' + (TASKS.length + 1) + '" style="text-align:center;padding:32px;color:#999">事業者がいません</td></tr>';

      }

      document.getElementById('tableBody').innerHTML = bodyHtml;

    }


    function cycleStatus(name, task, el) {

      var idx = STATUSES.indexOf(el.textContent);

      var next = STATUSES[(idx + 1) % STATUSES.length];

      el.textContent = next;

      el.className = 'badge badge-' + next;


      var company = allData.find(function(c) { return c.name === name; });

      if (company) company.tasks[task] = next;

      renderStats();

      if (!showCompleted && isAllDone(company)) renderTable();


      saveData().catch(function(e) { alert('保存失敗: ' + e.message); });

      updateTime();

    }


    function toggleCompleted() {

      showCompleted = !showCompleted;

      document.getElementById('toggleBtn').textContent = showCompleted ? '完了済みを非表示' : '完了済みを表示';

      renderTable();

    }


    function openAddModal() {

      document.getElementById('newCompanyName').value = '';

      document.getElementById('addModal').classList.add('show');

      setTimeout(function() { document.getElementById('newCompanyName').focus(); }, 100);

    }


    function closeAddModal() { document.getElementById('addModal').classList.remove('show'); }


    function addCompany() {

      var name = document.getElementById('newCompanyName').value.trim();

      if (!name) { alert('事業者名を入力してください'); return; }

      if (allData.some(function(c) { return c.name === name; })) { alert('すでに存在します'); return; }


      var newC = { name: name, tasks: {} };

      TASKS.forEach(function(t) { newC.tasks[t] = '未着手'; });

      allData.push(newC);

      render();

      closeAddModal();

      saveData().catch(function(e) { alert('保存失敗: ' + e.message); });

    }


    function updateTime() {

      document.getElementById('lastUpdated').textContent = '最終更新: ' + new Date().toLocaleTimeString('ja-JP');

    }


    function esc(s) { return String(s).replace(/&/g,'&amp;').replace(/</g,'&lt;').replace(/>/g,'&gt;'); }

    function escAttr(s) { return String(s).replace(/\\/g,'\\\\').replace(/'/g,"\\'"); }


    document.getElementById('newCompanyName').addEventListener('keydown', function(e) {

      if (e.key === 'Enter') addCompany();

      if (e.key === 'Escape') closeAddModal();

    });


    document.getElementById('tokenInput').addEventListener('keydown', function(e) {

      if (e.key === 'Enter') saveToken();

    });


    init();

  </script>
